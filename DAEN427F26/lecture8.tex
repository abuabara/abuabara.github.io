\begin{frame}{Learning goals}
By the end, you should be able to:
\begin{enumerate}
\item Summarize a posterior using a mean and credible interval.
\item Calculate the probability of a decision-relevant claim.
\item Predict future observations, including uncertainty.
\item Compare actions using \textbf{expected loss} and a risk tolerance.
\item Explain how simulation supports these calculations.
\end{enumerate}
\medskip
\textbf{Big idea:} Bayesian analysis does not end with an estimate. It helps us \emph{choose an action under uncertainty}.
\end{frame}

\begin{frame}{Running example from Chapter 8: museum artists}
Let $\pi$ be the proportion of artists in major US modern-art museums who were born in 1965 or later.
\begin{itemize}
\item Prior: $\pi\sim\operatorname{Beta}(4,6)$.
\item Sample: 14 of 100 artists in a MoMA sample are in this group.
\item Sampling model: $Y\given\pi\sim\operatorname{Binomial}(100,\pi)$.
\end{itemize}
\[
\boxed{\pi\given Y=14\sim\operatorname{Beta}(4+14,\,6+86)
=\operatorname{Beta}(18,92)}
\]
\textbf{Interpretation:} after seeing the data, smaller proportions are more plausible than the prior suggested.
\end{frame}

\begin{frame}{A posterior is more than one number}
For $\pi\given Y=14\sim\operatorname{Beta}(18,92)$:
\begin{itemize}
\item Posterior mean: $\E(\pi\given Y)=18/110\approx\mathbf{0.164}$.
\item Posterior mode: $(18-1)/(110-2)\approx0.157$.
\item Middle 95\% credible interval: $\mathbf{(0.101,\,0.238)}$.
\end{itemize}
\medskip
\textbf{Plain language:} the proportion is likely near 16\%, but values around 10\% to 24\% remain plausible.
\medskip
\textbf{Risk lesson:} two estimates with the same mean can lead to different decisions if their uncertainty differs.
\end{frame}

\begin{frame}{Understanding credible intervals}
A 95\% Bayesian credible interval $[L,U]$ is a range satisfying
\[
\Prb(L\leq\pi\leq U\given Y)=0.95.
\]
For our example:
\[
\Prb(0.101\lesssim\pi\lesssim0.238\given Y=14)\approx0.95.
\]
\begin{block}{Do not confuse two questions}
\begin{itemize}
\item \textbf{Parameter uncertainty:} What proportion $\pi$ is plausible?
\item \textbf{Future variability:} What might happen in the \emph{next} sample?
\end{itemize}
\end{block}
A credible interval addresses the first question, not the second.
\end{frame}

\begin{frame}{Posterior probabilities answer useful questions}
Suppose a planning team asks:\medskip
\begin{center}
\emph{Is the true proportion below 20\%?}
\end{center}
For the museum posterior,
\[
\Prb(\pi<0.20\given Y=14)=\mathbf{0.849}.
\]
\begin{itemize}
\item The model assigns about 85\% probability to the claim.
\item The complementary probability is about 15\%.
\item This is \emph{not} a guarantee; choices also depend on consequences.
\end{itemize}
\textbf{Decision connection:} a threshold matters when it changes what we would do.
\end{frame}

\begin{frame}{Hypotheses and Bayes factors: the intuition}
Compare two claims:
\[
H_a:\pi<0.20,\qquad H_0:\pi\geq0.20.
\]
\begin{itemize}
\item Prior chance of $H_a$: $0.0856$.
\item Posterior chance of $H_a$: $0.849$.
\item Posterior odds: $0.849/0.151\approx5.62$.
\end{itemize}
\[
\text{Bayes factor}=\frac{\text{posterior odds}}{\text{prior odds}}\approx\mathbf{60}.
\]
The data shifted the odds strongly toward $H_a$.
\medskip
\textbf{Important:} a large Bayes factor is evidence, \emph{not} a decision rule; costs of mistakes still matter.
\end{frame}

\begin{frame}{A caution about exact-value hypotheses}
For a continuous parameter with a continuous posterior,
\[
\Prb(\pi=0.30\given Y)=0.
\]
That does \emph{not} automatically establish a useful decision.
\begin{itemize}
\item Instead, ask whether a value is \textbf{practically different}.
\item Example: ``roughly 30\%'' means $0.25\le\pi\le0.35$.
\item Calculate posterior probability of this \emph{range}.
\end{itemize}
\textbf{Decision lesson:} define a meaningful tolerance before looking at results.
\end{frame}

\begin{frame}{Prediction: what happens next?}
Suppose we examine 20 \emph{new} museum artists; $Y'$ is the number born in 1965 or later.
\[
Y'\given\pi\sim\operatorname{Binomial}(20,\pi).
\]
The expected future count is
\[
\E(Y'\given Y=14)=20\E(\pi\given Y=14)
=20(18/110)\approx\mathbf{3.27}.
\]
But observing exactly 3 is not certain!
\medskip
Prediction includes \textbf{sampling randomness} and \textbf{uncertainty about $\pi$}.
\end{frame}

\begin{frame}{Posterior predictive distribution}
To predict a future count, average over plausible values of $\pi$:
\[
\Prb(Y'=k\given Y)=\int_0^1\Prb(Y'=k\given\pi)\,p(\pi\given Y)\,d\pi.
\]
In the Beta--Binomial model this gives a \textbf{Beta--Binomial} predictive distribution.
\begin{itemize}
\item Predicted mean number of younger artists out of 20: $3.27$.
\item Chance of at least 5: $\Prb(Y'\ge5\given Y=14)\approx\mathbf{0.233}$.
\end{itemize}
\textbf{Risk lesson:} often the most important answer is a \emph{tail probability}, not the expected count.
\end{frame}

\begin{frame}{From inference to decision analysis}
A decision requires \textbf{three ingredients}:
\begin{enumerate}
\item \textbf{Actions:} What can we choose?
\item \textbf{Uncertain states/outcomes:} What could happen?
\item \textbf{Loss or payoff:} What is the consequence of each action and state?
\end{enumerate}
\medskip
For an action $a$, define its posterior expected loss:
\[
\boxed{R(a\given Y)=\E\big[L(a,\theta)\given Y\big].}
\]
Choose the action with the \textbf{smallest expected loss} if that is the agreed decision objective.
\end{frame}

\begin{frame}{New example: quality-control risk}
A manufacturer wants to estimate $\theta$, the probability that a new part fails inspection.
\begin{itemize}
\item Prior: $\theta\sim\operatorname{Beta}(2,18)$ (mean 10\%).
\item Inspect 32 parts; 3 fail and 29 pass.
\item Posterior: $\theta\given Y\sim\operatorname{Beta}(5,47)$.
\end{itemize}
\[
\E(\theta\given Y)=\frac5{52}\approx\mathbf{9.6\%}.
\]
95\% credible interval: approximately \textbf{3.3\% to 18.9\%}.
\medskip
\textbf{Management question:} Should we perform an extra screening step before shipment?
\end{frame}

\begin{frame}{Define the available actions and losses}
Suppose the team can either \textbf{ship as is} or \textbf{screen every part}. For teaching purposes, assume:
\begin{itemize}
\item Shipping an unchecked failed part costs $\$200$ on average.
\item Screening costs $\$12$ per part.
\item Screening identifies and resolves every failure at no further cost.
\end{itemize}
\medskip
For a future part,
\[
\begin{aligned}
\E(L_{\mathrm{ship}}\given Y)&=200\E(\theta\given Y),\\
\E(L_{\mathrm{screen}}\given Y)&=12.
\end{aligned}
\]
\textbf{Note:} These are simplified, illustrative cost assumptions, not real factory estimates.
\end{frame}

\begin{frame}{Compute expected loss and choose}
Under the posterior mean failure rate $5/52$:
\[
\begin{aligned}
R(\mathrm{ship}\given Y)&=200\times\frac5{52}\approx\$19.23,\\
R(\mathrm{screen}\given Y)&=\$12.00.
\end{aligned}
\]
\begin{block}{Decision under the stated assumptions}
Screen the part: lower posterior expected loss by about \textbf{\$7.23 per part}.
\end{block}
For 1,000 comparable future parts, the expected saving is about $\$7{,}230$.
\medskip
This result depends on both the posterior \emph{and} the assumed costs.
\end{frame}

\begin{frame}{Break-even thresholds are easy to communicate}
Screening is preferable when
\[
12<200\E(\theta\given Y).
\]
Equivalently,
\[
\boxed{\E(\theta\given Y)>0.06.}
\]
\begin{itemize}
\item If expected failure probability exceeds 6\%, screen.
\item Our estimate is 9.6\%, so screen.
\item If costs change, the \textbf{break-even threshold changes}.
\end{itemize}
\textbf{Sensitivity question:} what if screening costs $\$22$ rather than $\$12$?
\end{frame}

\begin{frame}{Probability of crossing a risk threshold}
The posterior can also answer:
\[
\Prb(\theta>0.10\given Y)=\Prb(\theta>0.10\given\operatorname{Beta}(5,47))\approx\mathbf{0.413}.
\]
There is about a \textbf{41\%} chance the failure rate exceeds 10\%.
\medskip
\begin{itemize}
\item This probability could trigger further inspection or monitoring.
\item It need not change the expected-loss decision automatically.
\item Different organizations may have different risk tolerances.
\end{itemize}
\end{frame}

\begin{frame}{Prediction for a batch: another form of risk}
Suppose 20 more parts will be shipped.
\[
K\given\theta\sim\operatorname{Binomial}(20,\theta),\qquad
\theta\given Y\sim\operatorname{Beta}(5,47).
\]
Using the posterior predictive distribution:
\[
\E(K\given Y)=20(5/52)\approx\mathbf{1.92},
\]
\[
\Prb(K\ge3\given Y)\approx\mathbf{0.306}.
\]
Thus the expected count is about 2, but there is still about a \textbf{31\% chance of 3 or more} failures.
\end{frame}

\begin{frame}{Expected loss vs. downside risk}
Both may matter, but they answer different questions.
\begin{center}
\begin{tabular}{p{3.0cm}p{5.0cm}}
\toprule
\textbf{Criterion} & \textbf{Question}\\
\midrule
Expected loss & Which action has the smallest average cost?\\
Tail probability & How likely is an unacceptably bad outcome?\\
Credible interval & How uncertain is the underlying rate?\\
Predictive distribution & How variable might future results be?\\
\bottomrule
\end{tabular}
\end{center}
\medskip
A robust decision report makes the \textbf{objective, threshold, and assumptions} explicit.
\end{frame}

\begin{frame}{What is the value of more information?}
Imagine we could inspect more parts before deciding on screening.
\begin{itemize}
\item New data might push the estimated failure rate \emph{above or below} the 6\% break-even point.
\item That could change the preferred action.
\item Collecting more data has a cost, so we should ask whether it might \emph{improve the decision enough} to justify that cost.
\end{itemize}
\medskip
\textbf{Core idea:} information is valuable when it can change an action and reduce expected loss.
\end{frame}

\begin{frame}{Simulation: inference when formulas are difficult}
In a simple Beta--Binomial model, we can calculate many answers exactly. In more complex models we often simulate from the posterior (e.g., MCMC).
\begin{enumerate}
\item Draw many plausible $\theta^{(1)},\ldots,\theta^{(S)}$ values.
\item Approximate $\E(\theta\given Y)$ by their average.
\item Approximate $\Prb(\theta>0.10\given Y)$ by the fraction over 0.10.
\item For each draw, simulate future failures $K^{(s)}$.
\item Approximate risks and expected losses using simulated outcomes.
\end{enumerate}
\textbf{Practical check:} MCMC results need convergence and sampling diagnostics.
\end{frame}

\begin{frame}[fragile]{Optional R: four questions in a few lines}
\small
\begin{verbatim}
# Museum example, Chapter 8
18 / (18 + 92)                 # posterior mean
qbeta(c(.025, .975), 18, 92)  # credible interval
pbeta(.20, 18, 92)            # P(pi < .20 | data)

# Quality-control example
qbeta(c(.025, .975), 5, 47)
pbeta(.10, 5, 47, lower.tail = FALSE)
200 * 5 / 52                  # expected loss: ship
12                            # expected loss: screen
\end{verbatim}
\footnotesize
For the predictive tail, use the self-contained code on the next slide (no extra packages).
\end{frame}

\begin{frame}[fragile]{Optional R: posterior predictive tail}
\small
\begin{verbatim}
# Beta-Binomial probability, using base R
pbb <- function(k, n, a, b) {
  exp(lchoose(n, k) + lbeta(a + k, b + n - k)
      - lbeta(a, b))
}

# Museum: P(at least 5 out of the next 20)
sum(sapply(5:20, pbb, n = 20, a = 18, b = 92))

# Parts: P(at least 3 failures out of the next 20)
sum(sapply(3:20, pbb, n = 20, a = 5, b = 47))
\end{verbatim}
\textbf{Interpretation:} add probabilities of all outcomes that meet the adverse-event threshold.
\end{frame}

\begin{frame}{Class exercise: would you screen?}
A new product has posterior failure probability
\[
\theta\given Y\sim\operatorname{Beta}(3,27),
\qquad\E(\theta\given Y)=0.10.
\]
Suppose inspection costs $\$15$ per unit, and an unchecked failure costs $\$100$.
\begin{enumerate}
\item What is the expected loss of shipping without inspection?
\item What is the expected loss of inspecting every unit?
\item Which action minimizes expected loss?
\item If an unchecked failure cost $\$250$, would your answer change?
\end{enumerate}
\end{frame}

\begin{frame}{Class exercise: discussion answers}
\begin{enumerate}
\item Ship: $100\times0.10=\mathbf{\$10}$ expected loss.
\item Inspect: $\mathbf{\$15}$ expected loss.
\item With a $\$100$ failure cost: \textbf{ship}, under the stated assumptions.
\item With a $\$250$ failure cost: ship costs $250\times0.10=\$25$; \textbf{inspect} is preferred.
\end{enumerate}
\medskip
\textbf{Takeaway:} the data and posterior are unchanged; the optimal action changes because the consequences change.
\end{frame}

\begin{frame}{Five questions to ask in any risk analysis}
\begin{enumerate}
\item What is uncertain, and what is the posterior distribution?
\item What threshold or future event actually matters?
\item Which actions are available?
\item What are the costs and benefits of each outcome?
\item Would additional information be worth collecting?
\end{enumerate}
\medskip
\centering\textbf{Posterior $\longrightarrow$ prediction $\longrightarrow$ loss $\longrightarrow$ decision}
\end{frame}
