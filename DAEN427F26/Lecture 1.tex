\begin{frame}
\frametitle{Risk and Uncertainty}
\hspace{0.5cm} Often a distinction is made between {\color{red} risk} and {\color{blue}  uncertainty}: 
\begin{itemize} 
\item A choice under {\color{red} risk} is made when the outcomes are random and the probabilities of outcomes are known.
\item A choice under {\color{blue}  uncertainty} is made when the outcome probabilities are unknown.
\end{itemize}
\end{frame}
