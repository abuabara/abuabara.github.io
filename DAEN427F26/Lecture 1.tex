\begin{frame}{}
A decision problem begins with a \textbf{choice to be made}\\
under certain \textbf{objectives} and \textbf{constraints}.\\
\begin{block}{Risk and Uncertainty}
\begin{itemize}
\item \textbf{Choice under risk}: Outcome probabilities are known\\(or can be estimated reliably).
\item \textbf{Choice under uncertainty}: Outcome probabilities are unknown (or difficult to estimate).
\end{itemize}
\end{block}
\end{frame}

% Risk = known probabilities
% Uncertainty = unknown probabilities

%------------------------------------------------
\begin{frame}{}
\begin{block}{Decision and Risk Analysis}
\begin{enumerate}
\item \textbf{Modeling Decisions}: Define the available choices or actions.
\item \textbf{Modeling Risk}: Analyze situations where outcome probabilities are known.
\item \textbf{Modeling Uncertainty}: Analyze situations where outcome probabilities are unknown (or difficult to estimate).
\item \textbf{Modeling Preferences}: Represent how decision makers value outcomes and trade-offs.
\end{enumerate}
\end{block}
\end{frame}

%------------------------------------------------
\begin{frame}{}
\begin{block}{Decision and Risk Analysis, Examples}
\begin{enumerate}
\item \textbf{Modeling Decisions}\\A student chooses between studying for an exam tonight or attending a social event.
\item \textbf{Modeling Risk}\\A company launches a new product knowing there is a 70\% chance of high demand and a 30\% chance of low demand.
\item \textbf{Modeling Uncertainty}\\A farmer decides what crop to plant without knowing how much rain will fall this season.
\item \textbf{Modeling Preferences}\\A traveler chooses between a cheaper flight with a long layover and a more expensive direct flight.
\end{enumerate}
\end{block}
\end{frame}

%------------------------------------------------
\begin{frame}{}
\begin{block}{Bayesian Modeling for Decision Analysis}
\begin{itemize}
\item Bayesian modeling provides a way to update beliefs as new information becomes available.
\item It combines:
\begin{enumerate}
\item \textbf{Prior beliefs}: what we know / assume before seeing new data.
\item \textbf{Evidence/data}: new information relevant to the decision.
\item \textbf{Posterior beliefs}: updated probabilities after incorporating the evidence.
\end{enumerate}
\item In decision analysis, Bayesian modeling helps decision makers revise risk estimates and make better choices under uncertainty.
\item In short, it supports learning from evidence and improving decisions over time.
\end{itemize}
\end{block}
\end{frame}

\begin{frame}{}
\begin{block}{Interpreting Probability}
\begin{itemize}
\item The frequentist philosophy is so named for its interpretation of probability as the long-run relative frequency of a repeatable event.
\item In the Bayesian philosophy, a probability measures the relative plausibility of an event.
\end{itemize}
\end{block}

\begin{itemize}
\item[] \textbf{Is this coin fair?}
\item \textbf{Frequentist:} ``If the coin were fair, how surprising would these flips be?''
\item \textbf{Bayesian:} ``Given these flips, how much should I believe the coin is fair?''
\end{itemize}

\end{frame}

%------------------------------------------------
\begin{frame}{}
\begin{block}{One-Time Events}
\begin{itemize}
\item Many important events are unrepeatable:

\begin{itemize}
\item[] Big Bang, elections, weather and climate change, ...
\end{itemize}

\item \textbf{Frequentist} probability can be awkward for one-time events because there is no long-run repetition.
\item \textbf{Bayesian} probability is more flexible:

\begin{itemize}
\item It expresses degree of belief.
\item It incorporates model, evidence, and uncertainty.
\end{itemize}
\end{itemize}
\end{block}

\begin{itemize}
\item A 0.9 probability of winning means the candidate is considered 9 times more likely to win than lose.
\end{itemize}

\end{frame}

%------------------------------------------------
\begin{frame}{}
\begin{block}{The Bayesian Balancing Act}
\begin{itemize}
\item Bayesian and frequentist approaches interpret the same evidence differently:
\begin{itemize}
\item \textbf{Frequentist:} Treats "10 out of 10" identically in all cases.
\item \textbf{Bayesian:} Plausibility on observed evidence given prior beliefs.
\end{itemize}
\end{itemize}
\end{block}

\begin{itemize}
\item[] Both achieve 10 out of 10 correct in testing:
\item Zuofu claims he can predict coin flips.
\item[] ... is doubtful, predicting coin flips is highly implausible.
\item Kavya claims she can distinguish natural / artificial sweeteners.
\item[] ... is more believable.
\end{itemize}

\end{frame}

%------------------------------------------------
\begin{frame}{}
\begin{block}{Ask Questions!}
\begin{itemize}
\item A frequentist analysis assesses the uncertainty of the observed data in light of an assumed hypothesis.
\item A Bayesian analysis assesses the uncertainty of the hypothesis in light of the observed data.
\end{itemize}
\end{block}

\textbf{Scenario:}
\begin{itemize}
    \item Population: 100 people
    \item 4 have the disease, 96 do not have the disease
        \begin{center}
        \begin{tabular}{lccc}
        \hline
         & \textbf{Test +} & \textbf{Test --} & \textbf{Total} \\
        \hline
        Disease & 3 & 1 & 4 \\
        No disease & 9 & 87 & 96 \\
        \hline
        Total & 12 & 88 & 100 \\
        \hline
        \end{tabular}
        \end{center}
\end{itemize}
\end{frame}


\begin{frame}{}
\textbf{Frequentist question:} $P(\text{Positive Test} \mid \text{No Disease})$
\begin{itemize}
    \item Among the 96 healthy people, 9 tested positive
    \item False positive rate: $\frac{9}{96} \approx 0.10$
    \item If a person does \emph{not} have the disease, there is about a 10\% chance of testing positive.
    \begin{itemize}
        \item Frequentist analysis focuses on: $P(\text{Data} \mid \text{Null Hypothesis})$
        \item A p-value is \emph{not} the probability that the hypothesis is true.\\ (common misunderstanding)
    \end{itemize}
\end{itemize}

\vspace{.2cm}

\textbf{Bayesian question:} $P(\text{Disease} \mid \text{Positive Test})$
\begin{itemize}
    \item Only 3 of the 12 positive tests are true positives.
    \item Probability of actually having the disease: $\frac{3}{12} = 0.25$
    \item Even with a positive test, there is only a 25\% chance of having the disease.
    \begin{itemize}
    \item $\text{Sensitivity}: P(+ \mid \text{Disease}) = \frac{3}{4} = 75\%$
	\item $\text{Specificity}: P(- \mid \text{No Disease}) = \frac{87}{96} \approx 90.6\%$
	\end{itemize}
\end{itemize}

\end{frame}

\begin{frame}{}
\begin{block}{The Meaning of Probability}

\textbf{Probability describes uncertainty.}
\vspace{0.2cm}

\begin{columns}[t]
\begin{column}{0.45\textwidth}
\textbf{Frequentist:}
\begin{itemize}
    \item Probability is\\a \textbf{long-run frequency}.
    \item Describes what happens over many repeated trials.
\end{itemize}
\end{column}

\begin{column}{0.45\textwidth}
\textbf{Bayesian:}
\begin{itemize}
    \item Probability is\\a \textbf{degree of belief}.
    \item Measures how plausible something is given current information.
\end{itemize}
\end{column}
\end{columns}

\vspace{0.4cm}
Bayesian thinking updates beliefs using evidence:
\[
\text{Posterior} \propto \text{Prior} \times \text{Likelihood}
\]
Probability is not only about randomness;\\
it is a mathematical language for learning under uncertainty.
\end{block}
\end{frame}

%------------------------------------------------
\begin{frame}{}
[a.] Let's suppose that the first bag from your flight to enter the luggage carousel indeed has the same shape, size, and color as yours.
     Is it yours? Explain (i.e., calculate the posterior probability that the 1st bag is yours).

    \textbf{Answer:}
    {\color{blue}
    \footnotesize
    Let \(Y\) = first bag is yours, and \(M\) = first bag matches.
    
    \[
    \begin{aligned}
    &P(Y)=\frac{1}{100}\\
    &\mathcal{L}(Y^c\mid M)=P(M\mid Y^c)=0.05\\
    &\mathcal{L}(Y\mid M)=P(M\mid Y)=1\\
    \end{aligned}
    \]
    
    \[
    \begin{aligned}
    P(Y\mid M)&=\frac{P(M\mid Y)P(Y)}{P(M)}
              =\frac{P(M\mid Y)P(Y)}
                     {P(M\mid Y)P(Y)+P(M\mid Y^c)P(Y^c)}=\\
    &=\frac{1\cdot 0.01}{1\cdot 0.01+0.05\cdot 0.99}
    =\frac{0.01}{0.0595}\approx 0.168
    \end{aligned}
    \]
    
    So even though the first bag matches, it is probably not yours.

    }
    \end{frame}

%------------------------------------------------
\begin{frame}{}
[b.] Now suppose that we continue to wait for our bag to appear, failing to see it among the first 98 bags to enter the carousel (i.e., only 2 bags are left to come off the airplane).
     Calculate the posterior probability that the 99th bag, which also matches ours in shape, size, and color, is our own.
     
    \textbf{Answer:}
    {\color{blue}
    \footnotesize
    \[
    \begin{aligned}
    P(Y\mid M)
    &=\frac{1\cdot \tfrac{1}{\text{total of bags remaining}}}
           {1\cdot \tfrac{1}{\text{total of bags remaining}} + 0.05\cdot \tfrac{\text{total of bags remaining} - 1}{\text{total of bags remaining}}}\\
    &\text{(if 2 bags remain)}\\
    &=\frac{1\cdot \tfrac{1}{2}}
           {1\cdot \tfrac{1}{2} + 0.05\cdot \tfrac{1}{2}}=\frac{1}{1+0.05} \approx 0.9524
    \end{aligned}
    \]
    
    There is a 95.24\% chance that the 99th bag is mine given that it looks like mine.
    }
\end{frame}
