%------------------------------------------------
\begin{frame}{}
\begin{block}{Categorical data}
\vocab{Nominal data}
\begin{enumerate}
\item Named categories with no inherent order or numeric meaning.
\item When there are only two categories, the variable is often called binary or dichotomous
\item Examples: Material type (steel, aluminum, concrete), Model outcome (approved, rejected), Class label (spam, not spam)
\end{enumerate}
\vocab{Ordinal data}
\begin{enumerate}
\item Categories with a natural order, but where the spacing between levels is not necessarily measurable or equal.
\item Examples: Risk level (low, medium, high), Damage severity (minor, moderate, severe), Rating (poor, fair, good, excellent)
\end{enumerate}
\end{block}
\end{frame}

%------------------------------------------------
\begin{frame}{}
\begin{block}{Numeric data}
\vocab{Count or rank data}
\begin{enumerate}
\item (hopefully self-explanatory)
\item Examples: number of defects, number of clicks, number of missing values.
\end{enumerate}
\vocab{Continuous data}
\begin{enumerate}
\item Measurable quantities where difference between possible values can be arbitrarily small.
\item Data might lie within a range or be unbounded (in either direction)
\item Examples: temperature, pressure, time spent, latency, model error.
\end{enumerate}
\end{block}
\end{frame}

\begin{frame}{}
\begin{block}{}
\vocab{Probability spaces} are objects that model random experiments - real-world processes involving states that occur "randomly".

\vspace{0.2cm}
A probability space consists of three components:
\begin{enumerate}
  \item A \vocab{sample space}, containing all possible outcomes.
  \item Subsets of the sample space, called \vocab{events}, which comprise any number of possible outcomes (including none of them!).
  \item A function that assigns \vocab{probabilities} to events.
\end{enumerate}

\vspace{0.2cm}
An event is said to \vocab{occur} if the outcome of the random experiment is contained in that event.
\end{block}
\end{frame}

\begin{block}{Probability}
Is a number between 0 and 1 assigned to an event.

\vspace{0.4cm}
\begin{itemize}
  \item $P(\text{heads in one fair coin toss}) = 0.5$
  \item $P(\text{at least one head in two fair coin tosses}) = 0.75$

\begin{frame}{}
\begin{block}{Events as subsets}
Remember, events are subsets of the entire sample space.
Let's take for now the example of tossing a single fair coin and recording the outcome.
For one fair coin toss, the sample space is $S = \{H,T\}$.

\vspace{0.2cm}
We can define the simple events $A$ and $B$ in the outcome space:
\begin{itemize}
  \item $A$: getting a head, $A = \{H\}$.
  \item $B$: getting a tail, $B = \{T\}$.
\end{itemize}

\vspace{0.2cm}
Other possible events include: $\emptyset, \{H,T\}$.
\emph{What do the empty event and the full sample space mean in words?}
\end{block}
\end{frame}

%------------------------------------------------
\begin{frame}{}
\begin{block}{Set operations for events}
For two events $A$ and $B$:

\begin{itemize}
  \item \vocab{Intersection}: $A \cap B$ means $A$ and $B$ both occur.
  \item \vocab{Union}: $A \cup B$ means $A$ or $B$ occurs, including both.
  \item \vocab{Complement}: $A^c$ means $A$ does not occur.
  \item \vocab{Set difference}: $A \setminus B$ means $A$ occurs but $B$ does not.
\end{itemize}

\vspace{0.3cm}
Events $A$ and $B$ are \vocab{disjoint} or \vocab{mutually exclusive} if
\[
A \cap B = \emptyset.
\]
\end{block}
\end{frame}

\begin{frame}{}
\begin{block}{Kolmogorov axioms}
A probability rule $P(\cdot)$ must satisfy three basic axioms.

\begin{enumerate}
  \item For every event $A$, $P(A) \geq 0$.
  \item The probability of the entire sample space is $P(S)=1$.
  \item If $A$ and $B$ are disjoint, then
  \[
  P(A\cup B)=P(A)+P(B).
  \]
\end{enumerate}

\vspace{0.2cm}
These axioms are enough to derive many useful probability rules.
\end{block}
\url{https://en.wikipedia.org/wiki/Probability_axioms}
\end{frame}

%------------------------------------------------
\begin{frame}{}
\begin{block}{Two useful rules ...}
From the axioms, we obtain two rules used constantly.

\vspace{0.4cm}
\vocab{Complement rule}
\[
P(A^c)=1-P(A).
\]

\vocab{Inclusion-exclusion}
\[
P(A\cup B)=P(A)+P(B)-P(A\cap B).
\]
\end{block}
\end{frame}

%------------------------------------------------
\begin{frame}{}
\begin{block}{Why subtract the intersection?}
\vspace{0.4cm}
\centering
\begin{tikzpicture}[scale=0.9]
  \fill[AccentBlue!25] (-0.7,0) circle (1.5);
  \fill[AccentBlue!25] (0.7,0) circle (1.5);
  \begin{scope}
    \clip (-0.7,0) circle (1.5);
    \fill[AggieMaroon!40] (0.7,0) circle (1.5);
  \end{scope}
  \draw[thick] (-0.7,0) circle (1.5);
  \draw[thick] (0.7,0) circle (1.5);
  \node at (-1.5,0) {$A$};
  \node at (1.5,0) {$B$};
  \node at (0,-0.1) {$A\cap B$};
\end{tikzpicture}

\vspace{0.4cm}
When we add $P(A)+P(B)$, the overlap is counted twice. Inclusion-exclusion subtracts one copy of the overlap.
\end{block}
\end{frame}

%------------------------------------------------
\begin{frame}{}
\begin{block}{De Morgan's laws}
Complements, unions, and intersections can be combined:

\vspace{0.2cm}
\vocab{Complement of a union}
\[
(A\cup B)^c = A^c \cap B^c
\]

\vspace{0.2cm}
\vocab{Complement of an intersection}
\[
(A\cap B)^c = A^c \cup B^c
\]
\end{block}

The rules can be expressed in English as:
\begin{itemize}
  \item The negation of "A or B" is the same as "not A and not B".
  \item The negation of "A and B" is the same as "not A or not B".
\end{itemize}
%{\footnotesize \url{https://en.wikipedia.org/wiki/De_Morgan\%27s_laws}}
\end{frame}

%------------------------------------------------
\begin{frame}{}
\begin{block}{Example: engineering reliability}
A manufacturing line classifies parts using two screening tests.

\begin{itemize}
  \item $A$: a part fails the visual inspection.
  \item $B$: a part fails the sensor-based inspection.
\end{itemize}

Suppose
\[
P(A)=0.08, \qquad P(B)=0.12, \qquad P(A\cap B)=0.03.
\]

\vspace{0.2cm}
Then the probability that a part fails at least one inspection is
\[
P(A\cup B)=0.08+0.12-0.03=0.17.
\]
\end{block}
\end{frame}

%------------------------------------------------
\begin{frame}{}
\begin{block}{Example: data science classification}
A model flags observations for review.

\begin{itemize}
  \item $A$: the observation is truly anomalous.
  \item $B$: the model flags the observation.
\end{itemize}

Important event statements include:
\begin{itemize}
  \item $A\cap B$: true positive.
  \item $A^c\cap B$: false positive.
  \item $A\cap B^c$: false negative.
  \item $A^c\cap B^c$: true negative.
\end{itemize}
\end{block}
\end{frame}

\begin{frame}{}
\begin{block}{Bayesian Modeling of\\Decisions, Uncertainty, Risk, and Preferences}
\begin{itemize}
\item Bayesian modeling provides a way to update beliefs as new information becomes available.
\item It combines:
\begin{enumerate}
\item \vocab{Prior beliefs}: what we know / assume before seeing new data.
\item \vocab{Evidence/data}: new information relevant to the decision.
\item \vocab{Posterior beliefs}: updated probabilities after incorporating the evidence.
\end{enumerate}
\item Bayesian modeling helps decision makers revise risk estimates and make better choices under uncertainty.
\item Supporting learning from evidence and improving decisions over time.
\end{itemize}
\end{block}
\end{frame}

\begin{frame}{}
\begin{block}{Interpreting Probability}
\begin{itemize}
\item The frequentist philosophy is so named for its interpretation of probability as the long-run relative frequency of a repeatable event.
\item In the Bayesian philosophy, a probability measures the relative plausibility of an event.
\end{itemize}
\end{block}

\begin{itemize}
\item[] \vocab{Is this coin fair?}
\item \vocab{Frequentist:} ``If the coin were fair, how surprising would these flips be?''
\item \vocab{Bayesian:} ``Given these flips, how much should I believe the coin is fair?''
\end{itemize}

\end{frame}

%------------------------------------------------
\begin{frame}{}
\begin{block}{One-Time Events}
\begin{itemize}
\item Many important events are unrepeatable:

\begin{itemize}
\item[] Big Bang, elections, weather and climate change, ...
\end{itemize}

\item \vocab{Frequentist} probability can be awkward for one-time events because there is no long-run repetition.
\item \vocab{Bayesian} probability is more flexible:

\begin{itemize}
\item It expresses degree of belief.
\item It incorporates model, evidence, and uncertainty.
\end{itemize}
\end{itemize}
\end{block}

\begin{itemize}
\item A 0.9 probability of winning means the candidate is considered 9 times more likely to win than lose.
\end{itemize}

\end{frame}

%------------------------------------------------
\begin{frame}{}
\begin{block}{The Bayesian Balancing Act}
\begin{itemize}
\item Bayesian and frequentist approaches interpret the same evidence differently:
\begin{itemize}
\item \vocab{Frequentist:} Treats "10 out of 10" identically in all cases.
\item \vocab{Bayesian:} Plausibility on observed evidence given prior beliefs.
\end{itemize}
\end{itemize}
\end{block}

\begin{itemize}
\item[] Both achieve 10 out of 10 correct in testing:
\item Zuofu claims he can predict coin flips.
\item[] ... is doubtful, predicting coin flips is highly implausible.
\item Kavya claims she can distinguish natural / artificial sweeteners.
\item[] ... is more believable.
\end{itemize}

\end{frame}

\begin{frame}{}
\begin{block}{Ask Questions!}
\begin{itemize}
\item A \vocab{frequentist} analysis assesses the uncertainty of the observed data in light of an assumed hypothesis.
\item A \vocab{Bayesian} analysis assesses the uncertainty of the hypothesis in light of the observed data.
\end{itemize}
\end{block}

\vspace{.2cm}

\vocab{Scenario:} Population 100 people, 4 have the disease, 96 do not.
        \begin{center}
        \begin{tabular}{lccc}
        \hline
         & \vocab{Test +} & \vocab{Test --} & \vocab{Total} \\
        \hline
        Disease & 3 & 1 & 4 \\
        No disease & 9 & 87 & 96 \\
        \hline
        Total & 12 & 88 & 100 \\
        \hline
        \end{tabular}
        \end{center}
\end{frame}


\begin{frame}{}
\vocab{Frequentist question:} $P(\text{Positive Test} \mid \text{No Disease})$
\begin{itemize}
    \item Among the 96 healthy people, 9 tested positive
    \begin{itemize}
        \item False positive rate: $\frac{9}{96} \approx 0.10$
        \item[] If a person does \emph{not} have the disease,\\there is about a 10\% chance of testing positive.
    \end{itemize}
    \item Frequentist analysis focuses on: $P(\text{Data} \mid \text{Null Hypothesis})$
    \item A p-value is \emph{not} the probability that the hypothesis is true.\\ (common misunderstanding)
\end{itemize}

\vspace{.2cm}

\vocab{Bayesian question:} $P(\text{Disease} \mid \text{Positive Test})$
\begin{itemize}
    \item Only 3 of the 12 positive tests are true positives.
    \begin{itemize}
        \item Probability of actually having the disease: $\frac{3}{12} = 0.25$
        \item[] Even with a positive test, disease probability is only 25\%.
	\end{itemize}
    \item $\text{Sensitivity}: P(+ \mid \text{Disease}) = \frac{3}{4} = 75\%$
	\item $\text{Specificity}: P(- \mid \text{No Disease}) = \frac{87}{96} \approx 90.6\%$
\end{itemize}

\end{frame}

\begin{frame}{}
\begin{block}{The Meaning of Probability}

\vocab{Probability describes uncertainty.}
\vspace{0.2cm}

\begin{columns}[t]
\begin{column}{0.45\textwidth}
\vocab{Frequentist:}
\begin{itemize}
    \item Probability is\\a \vocab{long-run frequency}.
    \item Describes what happens over many repeated trials.
\end{itemize}
\end{column}

\begin{column}{0.45\textwidth}
\vocab{Bayesian:}
\begin{itemize}
    \item Probability is\\a \vocab{degree of belief}.
    \item Measures how plausible something is given current information.
\end{itemize}
\end{column}
\end{columns}

\vspace{0.4cm}
Bayesian thinking updates beliefs using evidence:
\[
\color{aggiemaroon}{
\text{\vocab{Posterior}} \propto \text{\vocab{Prior}} \times \text{\vocab{Likelihood}}
}
\]
Probability is not only about \ul{randomness};\\
it is a mathematical language for learning under \hl{uncertainty}.
\end{block}
