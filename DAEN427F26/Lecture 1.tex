\begin{frame}{}
A decision problem begins with a \textbf{choice to be made}\\
under certain \textbf{objectives} and \textbf{constraints}.\\
\begin{block}{Risk and Uncertainty}
\begin{itemize}
\item \textbf{Choice under risk}: Outcome probabilities are known\\(or can be estimated reliably).
\item \textbf{Choice under uncertainty}: Outcome probabilities are unknown (or difficult to estimate).
\end{itemize}
\end{block}
\end{frame}

% Risk = known probabilities
% Uncertainty = unknown probabilities

%------------------------------------------------
\begin{frame}{}
\begin{block}{Decision and Risk Analysis}
\begin{enumerate}
\item \textbf{Modeling Decisions}: Define the available choices or actions.
\item \textbf{Modeling Risk}: Analyze situations where outcome probabilities are known.
\item \textbf{Modeling Uncertainty}: Analyze situations where outcome probabilities are unknown (or difficult to estimate).
\item \textbf{Modeling Preferences}: Represent how decision makers value outcomes and trade-offs.
\end{enumerate}
\end{block}
\end{frame}

%------------------------------------------------
\begin{frame}{}
\begin{block}{Bayesian Modeling for Decision Analysis}
\begin{itemize}
\item Bayesian modeling provides a way to update beliefs as new information becomes available.
\item It combines:
\begin{enumerate}
\item \textbf{Prior beliefs}: what we know or assume before seeing new data.
\item \textbf{Evidence/data}: new information relevant to the decision.
\item \textbf{Posterior beliefs}: updated probabilities after incorporating the evidence.
\end{enumerate}
\item In decision analysis, Bayesian modeling helps decision makers revise risk estimates and make better choices under uncertainty.
\item In short, it supports learning from evidence and improving decisions over time.
\end{itemize}
\end{block}
\end{frame}
