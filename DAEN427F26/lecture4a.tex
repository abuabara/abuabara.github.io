\begin{frame}{Learning objectives}
By the end of class, you should be able to:
\begin{itemize}
  \item explain how prior information and observed data jointly determine a posterior;
  \item quantify their influence in a Beta--Binomial model;
  \item update beliefs sequentially as evidence arrives;
  \item recognize when a prior prevents learning from evidence;
  \item connect posterior probabilities to risk decisions.
\end{itemize}
\end{frame}

\begin{frame}{Decision setting: an uncertain event rate}
Let $\pi$ denote the unknown probability of an event.
\begin{itemize}
  \item Safety: probability that a component fails during a mission.
  \item Operations: probability that demand exceeds capacity.
  \item Screening: probability that a sampled item is nonconforming.
\end{itemize}
We observe $Y$ events in $n$ comparable trials:
\[
Y\mid \pi \sim \operatorname{Bin}(n,\pi), \qquad
\pi\sim\operatorname{Beta}(\alpha,\beta).
\]
\riskbox{The posterior describes uncertainty about the event rate. A decision still requires consequences, alternatives, and a decision rule.}
\end{frame}

\begin{frame}{Reading Beta priors}
For $\pi\sim\operatorname{Beta}(\alpha,\beta)$:
\[
E(\pi)=\frac{\alpha}{\alpha+\beta}, \qquad
\operatorname{Var}(\pi)=\frac{\alpha\beta}{(\alpha+\beta)^2(\alpha+\beta+1)}.
\]
\begin{itemize}
  \item The ratio $\alpha:(\alpha+\beta)$ locates the prior mean.
  \item The total $\alpha+\beta$ measures prior concentration and behaves like a prior sample size.
  \item A vague prior spreads plausibility broadly; an informative prior concentrates it.
\end{itemize}
\riskbox{Prior concentration determines how much new evidence a model needs before a risk estimate moves substantially.}
\end{frame}

\begin{frame}[fragile]{R: inspect competing priors}
\begin{lstlisting}

\end{lstlisting}
\textbf{Interpretation (before computation!):}\\
location represents belief; spread represents certainty.
\end{frame}

\begin{frame}{Conjugate updating keeps the calculation simple}
After observing $Y=y$ successes in $n$ trials,
\[
\pi\mid(Y=y)\sim\operatorname{Beta}(\alpha+y,\;\beta+n-y).
\]
The update has a useful counting interpretation:
\begin{align*}
\text{posterior success parameter} &= \alpha + \text{observed successes},\\
\text{posterior failure parameter} &= \beta + \text{observed failures}.
\end{align*}
\textbf{Note:} $\alpha$ and $\beta$ are hyperparameters.\\
Calling them literal past observations is often only an analogy.
\end{frame}

\begin{frame}{Posterior mean as a weighted compromise}
\[
E(\pi\mid y)=\frac{\alpha+y}{\alpha+\beta+n}
=\underbrace{\frac{\alpha+\beta}{\alpha+\beta+n}}_{w_{\text{prior}}}
\underbrace{\frac{\alpha}{\alpha+\beta}}_{\text{prior mean}}
+\underbrace{\frac{n}{\alpha+\beta+n}}_{w_{\text{data}}}
\underbrace{\frac{y}{n}}_{\text{sample rate}}.
\]
\[
w_{\text{prior}}+w_{\text{data}}=1.
\]
This identity makes the balance explicit. The prior contributes $\alpha+\beta$ units of concentration; the data contribute $n$ trials.
\end{frame}

\begin{frame}{Worked balance calculation}
Use the optimistic prior $\mathrm{Beta}(14,1)$.
\begin{columns}[T]
\column{.48\textwidth}
\textbf{Small sample: $6/13$}
\[
w_{\text{prior}}=\frac{15}{28}=0.536
\]
\footnotesize
\[
\begin{aligned}
E(\pi\mid y)
  &= 0.536(0.933)+0.464(0.462) \\
  &= 0.714.
\end{aligned}
\]
\column{.48\textwidth}
\textbf{Larger sample: $46/99$}
\[
w_{\text{prior}}=\frac{15}{114}=0.132
\]
\footnotesize
\[
\begin{aligned}
E(\pi\mid y)
  &= 0.132(0.933)+0.868(0.465) \\
  &= 0.526.
\end{aligned}
\]
\end{columns}
\vspace{.7em}
The same prior and almost the same observed rate lead to different posterior means because the evidence has different weight.
\end{frame}

\begin{frame}{What happens as evidence accumulates?}
For a fixed proper Beta prior,
\[
\frac{\alpha+\beta}{\alpha+\beta+n}\longrightarrow 0,
\qquad
\frac{n}{\alpha+\beta+n}\longrightarrow 1.
\]
Therefore,
\[
E(\pi\mid y)\longrightarrow \frac{y}{n}
\quad\text{as }n\text{ grows}.
\]
Analysts with different non-stubborn priors tend to converge when they observe strong common evidence.

Does convergence guarantee a good decision if the trials are biased or not comparable to the decision context?
\end{frame}

\begin{frame}
\begin{itemize}
\item More data do not automatically mean better inference.
\item If the observations are biased, dependent, measured incorrectly, or drawn from a population that differs from the decision population, the posterior may become extremely certain about the wrong value.
\item Bayesian updating correctly processes the evidence described by the model; it cannot by itself correct a flawed sampling or data-generating process.
\end{itemize}
\end{frame}

\begin{frame}{From probability to a risk decision}
Suppose action $a_1$ mitigates an adverse event and action $a_0$ does not. Let $L(a,\pi)$ denote loss.
\[
\text{Posterior expected loss}(a)=E[L(a,\pi)\mid y].
\]
Choose the action with smaller posterior expected loss.
\begin{block}{Simple threshold case}
If mitigation costs $C$ and an unmitigated event causes loss $D$, mitigate when
\[
P(\text{event}\mid y)\,D > C.
\]
\end{block}
Prior sensitivity matters most when competing posterior decisions lie near the action threshold.
\end{frame}

\begin{frame}{Sequential Bayesian learning}
As each batch arrives, yesterday's posterior becomes today's prior.
\[
\mathrm{Beta}(\alpha_{t-1},\beta_{t-1})
\xrightarrow{\ y_t\text{ of }n_t\ }
\mathrm{Beta}(\alpha_{t-1}+y_t,\beta_{t-1}+n_t-y_t).
\]
This supports risk monitoring in settings where evidence arrives over time: reliability tests, incident counts, inspections, and demand observations.
\end{frame}

\begin{frame}{Milgram example: three sequential batches}
The chapter begins with $\pi\sim\mathrm{Beta}(1,10)$ and updates three batches.
\begin{center}
\begin{tabular}{cccc}
\toprule
Stage & New data & Current model & Posterior mean \\
\midrule
0 & none & $\mathrm{Beta}(1,10)$ & 0.091 \\
1 & $1/10$ & $\mathrm{Beta}(2,19)$ & 0.095 \\
2 & $17/20$ & $\mathrm{Beta}(19,22)$ & 0.463 \\
3 & $8/10$ & $\mathrm{Beta}(27,24)$ & 0.529 \\
\bottomrule
\end{tabular}
\end{center}
The second batch causes the largest change because it conflicts sharply with the belief accumulated through stage 1.
\end{frame}

\begin{frame}[fragile]{R: implement sequential updating}
\begin{lstlisting}
batches <- tibble(day = 1:3,
                  y = c(1, 17, 8),
                  n = c(10, 20, 10))

alpha <- 1; beta <- 10
history <- tibble(day = 0, alpha = alpha, beta = beta)

for (i in seq_len(nrow(batches))) {
  alpha <- alpha + batches$y[i]
  beta  <- beta + batches$n[i] - batches$y[i]
  history <- rbind(history,
                   tibble(day = i, alpha = alpha, beta = beta))
}

history |> mutate(mean = alpha / (alpha + beta))
\end{lstlisting}
This pattern generalizes to any stream of Binomial batches.
\end{frame}

\begin{frame}{Order changes the path, not the destination}
Original order:
\[
\mathrm{Beta}(1,10)\to\mathrm{Beta}(2,19)\to
\mathrm{Beta}(19,22)\to\boxed{\mathrm{Beta}(27,24)}.
\]
Reverse order:
\[
\mathrm{Beta}(1,10)\to\mathrm{Beta}(9,12)\to
\mathrm{Beta}(26,15)\to\boxed{\mathrm{Beta}(27,24)}.
\]
The intermediate beliefs differ. The final posterior depends only on cumulative totals: 26 successes and 14 failures.
\riskbox{Operational actions taken between batches can differ even when the final statistical model is order invariant. Timing still matters for decisions.}
\end{frame}

\begin{frame}{Why order invariance holds}
For conditionally independent observations $y_1,y_2$,
\[
f(\theta\mid y_1,y_2)\propto f(\theta)L(\theta\mid y_1)L(\theta\mid y_2).
\]
Multiplication commutes:
\[
L(\theta\mid y_1)L(\theta\mid y_2)
=L(\theta\mid y_2)L(\theta\mid y_1).
\]
Sequential and pooled updates therefore yield the same final posterior under the same model assumptions.
\textbf{Boundary of the result:} dependence, changing data-generating processes, or adaptive sampling may require an expanded model.
\end{frame}

\begin{frame}{A prior can block learning}
Suppose a researcher assigns
\[
\pi\sim\mathrm{Unif}(0,0.25).
\]
Even after observing 8 events in 10 trials, Bayes' rule gives
\[
f(\pi\mid y=8)\propto \pi^8(1-\pi)^2
\quad\text{only for }0\leq\pi\leq0.25.
\]
The posterior cannot place probability above 0.25 because the prior assigns that region probability zero.
\begin{alertblock}{Modeling rule}
Avoid excluding technically possible parameter values unless the exclusion follows from a genuine physical or logical constraint.
\end{alertblock}
\end{frame}

\begin{frame}{Subjectivity and transparent analysis}
Bayesian analysis makes several judgments visible:
\begin{itemize}
  \item prior location and concentration;
  \item likelihood and sampling assumptions;
  \item outcomes, losses, and decision thresholds.
\end{itemize}
Good practice for risk analysis:
\begin{enumerate}
  \item justify the prior from data, expertise, or a stated weak-information goal;
  \item report sensitivity to plausible alternatives;
  \item inspect prior predictive implications;
  \item revise the model when the evidence reveals misspecification.
\end{enumerate}
\end{frame}

\begin{frame}{In-class (1)}
A maintenance action costs \$12,000. Failure costs \$100,000. Assume maintenance completely prevents failure and has no other consequence.

Let $p=P(\text{failure}\mid y)$. Compare posterior expected costs:
\[
\begin{aligned}
E[C\mid\text{no maintenance},y] &= 100{,}000p,\\
E[C\mid\text{maintenance},y] &= 12{,}000.
\end{aligned}
\]
Maintenance has lower expected cost when
\[
100{,}000p>12{,}000
\quad\Longleftrightarrow\quad
\boxed{p>0.12}.
\]
At $p=0.12$, the two actions have equal expected cost.
\end{frame}

\textbf{What the simple model omits}
\begin{itemize}
  \item Maintenance may reduce rather than eliminate failure risk.
  \item Failure losses and maintenance effectiveness may be uncertain.
  \item Downtime, safety, environmental, and reputational consequences may matter.
  \item Risk aversion and the value of waiting for more information may change the choice.
\end{itemize}
\end{frame}


\begin{frame}{Conclusions}
\begin{itemize}
  \item The posterior balances the prior and evidence.
  \item Prior concentration and sample size determine the balance in the Beta--Binomial model.
  \item Sequential updating reuses the current posterior as the next prior.
  \item The final posterior is order invariant under the stated model assumptions.
  \item Zero prior support cannot be repaired by contradictory data.
  \item Risk decisions require posterior uncertainty plus explicit consequences.
\end{itemize}
\end{frame}
