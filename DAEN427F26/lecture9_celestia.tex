% ----- Slide 3: Python code -----
\begin{frame}[fragile]{Python | A simple example}
  A short Python example using a list and its mean.

  \begin{code*}{python}
values = [4, 8, 12, 16]
mean_value = sum(values) / len(values)
print(f"Mean: {mean_value:.1f}")
  \end{code*}

  \textbf{Expected output:} \texttt{Mean: 10.0}
\end{frame}

% ----- Slide 4: R code -----
\begin{frame}[fragile]{R | The same calculation}
  The equivalent operation in base R.

  \begin{code*}{R}
values <- c(4, 8, 12, 16)
mean_value <- mean(values)
print(mean_value)
  \end{code*}

  \textbf{Expected output:} \texttt{[1] 10}
\end{frame}

% ----- Slide 5: Comparison -----
