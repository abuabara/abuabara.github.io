\begin{frame}{}
We want to understand how MCMC can generate samples from a posterior distribution.

\vspace{0.4cm}
Suppose we observe four Bernoulli trials:
\[
 y = (1,1,1,0)
\]

That is:
\[
 3 \text{ successes}, \qquad 1 \text{ failure}
\]

\vspace{0.3cm}
The unknown parameter is:
\[
\theta = \Pr(y=1)
\]
\end{frame}

\begin{frame}{}
%\textbf{Question:} We observe 3 successes in 4 trials.
What can we learn about the unknown success probability $\theta$?

\vspace{0.2cm}

\textbf{1. Prior:} All probabilities between 0 and 1
are initially equally plausible.
\[
\theta \sim \operatorname{Beta}(1,1)
\]

\textbf{2. Data and likelihood:}
\[
Y \mid \theta \sim \operatorname{Binomial}(4,\theta),
\qquad y=3
\]

\textbf{3. Bayesian updating:}
\[
\begin{aligned}
p(\theta \mid y)
&\propto p(y \mid \theta)\,p(\theta)\\
&\propto \theta^3(1-\theta)\cdot 1
\end{aligned}
\]

\textbf{4. Posterior distribution:}
\[
\boxed{\theta \mid y \sim \operatorname{Beta}(4,2)}
\]

Before observing data, all success probabilities were equally
plausible. After observing 3 successes in 4 trials, larger
probabilities become more plausible, but uncertainty remains.

\end{frame}

\begin{frame}{Metropolis Rule (Symmetric Proposal)}
Start at some current value:
\[
\theta_t
\]

Propose a new value:
\[
\theta^*
\]

For a symmetric proposal, $q(\theta^*\mid\theta_t)=q(\theta_t\mid\theta^*)$,
compute the acceptance ratio:
\[
 r = \frac{f(\theta^*)}{f(\theta_t)}
\]

\href{https://www.bayesrulesbook.com/chapter-7\#why-the-algorithm-works}{\footnotesize \textcolor{gray}{(check \hl{\it why the algorithm works)}}}

Then compute:
\[
 \alpha = \min(1,r)
\]

Accept the proposed value \hl{if}:
\[
 u < \alpha, \qquad u \sim \text{Uniform}(0,1)
\]
\end{frame}

\begin{frame}{Step 1: Move from 0.50 to 0.60}
Start at:
\[
\theta_0 = 0.50
\]

Compute:
\[
 f(0.50)=0.50^3(1-0.50)=0.125 \times 0.50=0.0625
\]

Propose:
\[
\theta^*=0.60
\]

Compute:
\[
 f(0.60)=0.60^3(1-0.60)=0.216 \times 0.40=0.0864
\]

Acceptance ratio:
\[
 r=\frac{0.0864}{0.0625}=1.3824
\]

So:
\[
\alpha=1
\]

Accept:
\[
0.50 \rightarrow 0.60
\]
\end{frame}

\begin{frame}{Step 2: Move from 0.60 to 0.80}
Current value:
\[
\theta_t=0.60
\]

Propose:
\[
\theta^*=0.80
\]

Compute:
\[
 f(0.80)=0.80^3(1-0.80)=0.512 \times 0.20=0.1024
\]

Acceptance ratio:
\[
 r=\frac{0.1024}{0.0864}=1.185
\]

Thus:
\[
\alpha=1
\]

Accept:
\[
0.60 \rightarrow 0.80
\]
\end{frame}

\begin{frame}{Step 3: A Rejected Proposal}
\small
Current value:
\[
\theta_t=0.80
\]

Propose:
\[
\theta^*=0.95
\]

Compute:
\[
 f(0.95)=0.95^3(1-0.95)\approx 0.8574\times 0.05=0.0429
\]

Acceptance ratio:
\[
 r=\frac{0.0429}{0.1024}\approx 0.419
\]

So:
\[
\alpha=0.419
\]

Suppose we draw: {\tiny \it(random value 0-1)} % {\tiny(random value between $0-1$)}:
\[
u=0.70
\]

Since:
\[
0.70>0.419
\]

reject the proposal and stay at:
\[
\theta=0.80
\]
\end{frame}

\begin{frame}{Step 4: Move from 0.80 to 0.70}
Current value:
\[
\theta_t=0.80
\]

Propose:
\[
\theta^*=0.70
\]

Compute:
\[
 f(0.70)=0.70^3(1-0.70)=0.343 \times 0.30=0.1029
\]

Acceptance ratio:
\[
 r=\frac{0.1029}{0.1024}\approx1.005
\]

Thus:
\[
\alpha=1
\]

Accept:
\[
0.80 \rightarrow 0.70
\]
\end{frame}

\begin{frame}{Manual Chain}
The chain we recorded is:
\[
0.50,\;0.60,\;0.80,\;0.80,\;0.70
\]

\vspace{0.4cm}
Notice the repeated value:
\[
0.80,\;0.80
\]

This happens because a rejected proposal keeps the chain at the current state.

\vspace{0.4cm}
After many iterations, the histogram of the chain approximates the posterior:
\[
p(\theta \mid y)
\]
\end{frame}

\begin{frame}{Key Idea}
MCMC does not only search for the best value.

\vspace{0.4cm}
It constructs a chain whose long-run behavior matches the posterior distribution.

\vspace{0.4cm}
Regions with higher posterior probability are visited more often.

\vspace{0.4cm}
Regions with lower posterior probability are visited less often, but not necessarily never.

\vspace{0.4cm}
That is why MCMC provides uncertainty, not just a point estimate.
\end{frame}

\begin{frame}{} % Connection to Bayesian Regression
In the simple example, the unknown was one parameter:
\[
\theta
\]

In Bayesian linear regression, the unknowns are usually:
\[
(\beta_0,\beta_1,\sigma)
\]

MCMC samples from:
\[
p(\beta_0,\beta_1,\sigma \mid x,y)
\]

Each draw gives one plausible regression model:
\[
y = \beta_0 + \beta_1 x + \varepsilon
\]

The collection of draws gives the full posterior uncertainty.
\end{frame}


\begin{frame}{Monte Carlo: Draw Samples Directly}
\small
\begin{enumerate}
\item Draw many independent values $x_1,\ldots,x_N$ from the distribution we want to study, $\pi(x)$.
\item For each value, calculate the result we care about, $g(x_i)$.
\item Average the results: $\frac{1}{N}\sum_{i=1}^N g(x_i)$.
\end{enumerate}
\vspace{0.25cm}
\textbf{Advantage:} Easy to understand; each draw is independent.\\[0.15cm]
\textbf{Challenge:} It may be hard to draw directly from a complicated distribution (e.g., complex shapes, high dimensions).\\[0.25cm]
\textbf{Main difference:} Monte Carlo draws independent values directly; Metropolis--Hastings builds a chain in which each value depends on the previous one.
\end{frame}

\begin{frame}{Metropolis--Hastings: How It Works}
\small
\begin{enumerate}
\item Start with a current value $x_t$.
\item Suggest a new value $x^*$ using a \textbf{proposal distribution} $q(x^*\mid x_t)$.
\item Calculate how likely we are to accept it:
\[
\alpha=\min\!\left\{1,\frac{\pi(x^*)\,q(x_t\mid x^*)}{\pi(x_t)\,q(x^*\mid x_t)}\right\}.
\]
\item Draw $u$ between 0 and 1. If $u<\alpha$, move to $x^*$;\\otherwise, stay at $x_t$.
\item Repeat to create a chain of samples.
\end{enumerate}
\vspace{0.1cm}
\textbf{$\pi(x)$:} The distribution we want to sample.\\
\textbf{$q(y\mid x)$:} The chance (or density) of proposing $y$ when at $x$.\\
{\footnotesize \textcolor{gray!85}{We use a proposal distribution because knowing how likely a value is under a posterior distribution is not the same as knowing how to \ul{generate random values} from that posterior.}}
\end{frame}

\begin{frame}{} % Why Do the $q$ Terms Cancel?
\small
The Metropolis--Hastings acceptance rule includes a \textbf{proposal ratio}:
\[
\alpha=\min\!\left\{1,\frac{\pi(x^*)}{\pi(x_t)}\times
\underbrace{\frac{q(x_t\mid x^*)}{q(x^*\mid x_t)}}_{\text{proposal ratio}}\right\}.
\]
If the proposal is \textbf{symmetric}, moving from $x_t$ to $x^*$ is just as likely as proposing the reverse move:
\[
q(x_t\mid x^*)=q(x^*\mid x_t).
\]
So we divide two equal numbers: the proposal ratio is $1$.
\[
\boxed{\alpha=\min\!\left\{1,\frac{\pi(x^*)}{\pi(x_t)}\right\}}
\]
\textbf{Example:} If each direction has proposal density $0.2$, then $0.2/0.2=1$.\\[0.1cm]
This simpler rule is called the \textbf{Metropolis algorithm}.
\end{frame}

\begin{frame}{Metropolis vs. Metropolis--Hastings}
\small
\begin{itemize}
\item \textbf{Metropolis:} The proposal is symmetric: it is equally likely to suggest a move in either direction. The $q$ terms cancel, so only the target distribution appears in the acceptance ratio.
\item \textbf{Metropolis--Hastings:} Also works with \emph{asymmetric} proposals. A move in one direction may be more likely than the reverse, so the $q$ ratio must stay in the formula.
\end{itemize}
\vspace{0.3cm}
\textbf{Remember!!} Metropolis is a special case of Metropolis--Hastings.\\[0.25cm]
Our earlier numerical example used the simpler \hl{Metropolis} rule.
\end{frame}

\begin{frame}{Independence Sampling}
\small
An \textbf{independence sampler} is a type of Metropolis--Hastings algorithm.
\begin{itemize}
\item It suggests new values from the \textbf{same fixed proposal distribution}, no matter where the chain currently is:
\[
q(x^*\mid x_t)=q(x^*).
\]
\item The acceptance rule becomes
\[
\alpha=\min\!\left\{1,\frac{\pi(x^*)\,q(x_t)}{\pi(x_t)\,q(x^*)}\right\}.
\]
\item It is \textbf{not usually Metropolis}, because the forward and reverse proposal densities may differ.
\end{itemize}
\vspace{0.1cm}
\textbf{Important:} Even though proposals do not depend on the current state, accepted samples can still depend on earlier samples.
\end{frame}

\begin{frame}{Conclusion}
\begin{itemize}
    \setlength{\itemsep}{0.3cm}

    \item \textbf{The posterior distribution} is defined
    by Bayes' rule, combining the prior and likelihood.

    \item \textbf{MCMC} generates a sequence of dependent
    samples whose long-run distribution is the posterior.

    \item \textbf{Metropolis-Hastings} is one MCMC algorithm:
    \begin{enumerate}
        \item Propose a new value.
        \item Accept or reject it using the posterior.
        \item Repeat many times.
    \end{enumerate}

    \item \textbf{Monte Carlo} uses the resulting samples
    to approximate posterior means, credible intervals,
    and probabilities.
\end{itemize}

\vspace{0.2cm}
MCMC does not create the posterior distribution.
It generates samples that allow us to
\textbf{explore and summarize the posterior}.
\end{frame}
