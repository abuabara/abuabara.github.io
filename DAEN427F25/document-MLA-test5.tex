\subsection{Broken telephone}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
Consider the following simplified version of the broken telephone game.
A first player secretly communicates a message
\( m_1 \in \{0,1,2\} \)
to a second player with probabilities
\[
P(m_1=0)=P(m_1=1)=P(m_1=2)=\frac{1}{3}.
\]

The second player then secretly communicates a message
\( m_2 = m_1 + n \)
to a third (final) player, where
\[
P(n=0)=\frac{2}{3},
\qquad
P(n=+1)=\frac{1}{3}.
\]

\begin{enumerate}
\item What is the optimal Bayes classification of the message received by the last player?
\item What is the expected error probability?
\end{enumerate}

\subsection{Binary classification}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
Consider the following binary classification problem with losses
\( L(C, C_{\mathrm{pred}}) \):
\[
\begin{array}{c|cc}
       & C_{\mathrm{pred}} = +1 & C_{\mathrm{pred}} = -1 \\
\hline
C = +1 & 0   & 100 \\
C = -1 & 10  & 0
\end{array}
\]

Your prior is
\[
P(C=+1)=\frac{1}{10}.
\]

\begin{enumerate}
\item What is the optimal classifier (and expected loss) based only on the prior distribution?
\item Suppose you can run an experiment with binary outcomes
\( x \in \{0,1\} \)
with likelihoods
\[
P(x = 1 \mid C = +1) = \frac{3}{4},
\qquad
P(x = 1 \mid C = -1) = \frac{1}{20}.
\]
Find the classifier that minimizes expected loss.
Compute the optimal expected loss and compare with part (a).
\end{enumerate}

\subsection*{2) Picking a Pope}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

Let's assume there are $7.8 \cdot 10^8$ humans and there is only one Pope (Pope Leo XIV at the time of this writing).
If a human is picked at random from the entire human population, the chances of that human being the pope are $1$ in $7.8 \cdot 10^8$.
\[
p(\text{Pope} \mid \text{human}) = \frac{1}{7.8 \cdot 10^8}
\]

Because Pope are selected among humans.
We can assert that given someone is the Pope, there is 100\% chance he is a human.
So, $p(\text{human} \mid \text{Pope}) = 1$.

Regarding the animated series Futurama and the space Pope is a reptile.
We then have that $p(\text{Pope} \mid \text{human}) = 0$ and $p(\text{human} \mid \text{Pope}) = 0$.

\subsection*{3) Broken telephone}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{enumerate}
\item
Note that $m_2\in \{0,1,2,3\}$
and
\[
P(m_2=m_1 \mid m_1)=\frac{2}{3} \quad P(m_2=m_1+1 \mid m_1)=\frac{1}{3}
\]

Hence the posterior distribution is:
\[
P(m_1=0 \mid m_2)=
\begin{cases}
\frac{\frac{2}{3} \cdot \frac{1}{3}}{\frac{2}{3} \cdot \frac{1}{3} + 0 \cdot \frac{1}{3} + 0 \cdot \frac{1}{3}} = 1, \quad m_2 = 0\\
\frac{\frac{1}{3} \cdot \frac{1}{3}}{\frac{1}{3} \cdot \frac{1}{3} + \frac{2}{3} \cdot \frac{1}{3} + 0\cdot\frac{1}{3}} = \frac{1}{3},\quad m_2 = 1\\
\frac{0 \cdot \frac{1}{3}}{0 \cdot \frac{1}{3} + \frac{1}{3} \cdot \frac{1}{3} + \frac{2}{3} \cdot \frac{1}{3}} = 0,\quad m_2 = 2 \\
\frac{0 \cdot \frac{1}{3}}{0 \cdot \frac{1}{3} + 0\cdot \frac{1}{3} + \frac{1}{3} \cdot \frac{1}{3}} = 0,\quad m_2 = 3.
\end{cases}
\]

\[
P(m_1=1 \mid m_2)=
\begin{cases}
\frac{0 \cdot \frac{1}{3}}{0 \cdot \frac{1}{3} + \frac{1}{3} \cdot \frac{1}{3} + 0 \cdot \frac{1}{3}}=0, \quad m_2 = 0\\
\frac{\frac{2}{3} \cdot \frac{1}{3}}{\frac{2}{3} \cdot \frac{1}{3} + \frac{1}{3} \cdot \frac{1}{3}+0 \cdot \frac{1}{3}} = \frac{2}{3}, \quad m_2 = 1\\
\frac{\frac{1}{3} \cdot \frac{1}{3}}{\frac{1}{3} \cdot \frac{1}{3} + \frac{2}{3} \cdot \frac{1}{3}+0 \cdot \frac{1}{3}} = \frac{1}{3}, \quad m_2 = 2\\
\frac{0 \cdot \frac{1}{3}}{0 \cdot \frac{1}{2} + 0 \cdot \frac{1}{3} + \frac{1}{3} \cdot \frac{1}{3}} = 0, \quad m_2 = 3.
\end{cases}
\]
and $P(m_1=2 \mid m_2)=1-P(m_1=0 \mid m_2)-P(m_1=1 \mid m_2)$.

Hence,
\[
m_{\mathrm{1_{pred}}}
%\hat{m_1}
(m_2)=
\begin{cases}
0,\quad m_2 = 0\\
1,\quad m_2 = 1\\
2,\quad m_2=2 \\
2,\quad m_2=3. 
\end{cases} 
\]

\item
$
P(m_1=0 \mid m_2=1)P(m_2=1) + P(m_1=1 \mid m_2=2) P(m_2=2) = \frac{1}{3} \cdot \frac{1}{3} + \frac{1}{3} \cdot \frac{1}{3} = \frac{2}{9}
$

\end{enumerate}

\subsection*{4) Binary classification}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{enumerate}
\item
The a priori Bayes classifier is $C_{\mathrm{pred}_{MAP}}=-1$ because $P(C=+1)<P(C=-1)$.
However, this is not the classifier the minimizes expected loss since: 
\begin{align*}
\mathbb{E}[L(C,C_{\mathrm{pred}}&=+1)] = 10\, P(C=-1) = 9 \\
                                &< \mathbb{E}[L(C,C_{\mathrm{pred}}=-1)] = 100\, P(C=+1) = 10
\end{align*}


The optimal expected loss is $\mathbb{E}[L(C,C_{\mathrm{pred}}=+1)]=10 P(C=-1)=9$.

\item
The posterior distribution is:
\begin{equation*}
P(C=+1 \mid x)=\left\{ 
\begin{array}{lll}
\frac{\frac{3}{4} \cdot \frac{1}{10}}{\frac{3}{4} \cdot \frac{1}{10} + \frac{1}{20} \cdot \frac{9}{10}} = 0.625  &  & x = 1\\
                                                                                                                 &  &  \\ 
\frac{\frac{1}{4} \cdot \frac{1}{10}}{\frac{1}{4} \cdot \frac{1}{10} + \frac{19}{20} \cdot \frac{9}{10}} = 0.0284&  & x=0
\end{array}
\right.
\end{equation*}

Hence,
\begin{align*}
\mathbb{E}[L(C,&C_{\mathrm{pred}}=+1) \mid x=1]=10 P(C=-1 \mid x=1)=3.75 \\
&< \mathbb{E}[L(C,C_{\mathrm{pred}}=-1) \mid x=1]
=100 P(C=+1 \mid x=1)=62.5
\end{align*}
and
\begin{align*}
\mathbb{E}[L(C,&C_{\mathrm{pred}}=+1) \mid x=0]=10 P(C=-1 \mid x=0)=9.71\\
&> \mathbb{E}[L(C,C_{\mathrm{pred}}=-1) \mid x=0]
=100 P(C=+1 \mid x=0)=2.84
\end{align*}

Hence, the optimal classifier is:
\begin{equation*}
C_{\mathrm{pred}}(x)=\left\{
\begin{array}{rrr}
+1 &  & x=1 \\ 
-1 &  & x=0
\end{array}
\right. 
\end{equation*}

The expected loss under optimal classifier is:
\begin{align*}
\mathbb{E}[L(C,C_{\mathrm{pred}}(x))]&=10P(C=-1 \mid x=1)P(x=1)+\\
&\qquad \qquad 100P(C=+1 \mid x=0)P(x=0)\\
&=10(0.375)P(x=1)+100(0.0284)P(x=0)\\
&=10(0.375)(0.12)+100(0.0284)(0.88) = 2.93 < 9
\end{align*}

\end{enumerate}
