\begin{frame}{What is classification?}
\begin{itemize}
  \item \textbf{Supervised learning}\\inputs (features $X$) and known labels (target $y$).
  \item \textbf{Classification}\\predicts a category, not a continuous value.
  \item \textbf{Examples}\\email type, equipment fault category, flower species.
\end{itemize}
\end{frame}

  \item $y$: species label.
  \item Built into \texttt{scikit-learn} (no external download required).
\end{itemize}
\end{frame}

\begin{frame}{Training workflow}
\begin{enumerate}
  \item Load the dataset and inspect its shape and classes.
  \item Split into training and test sets (e.g., 80\% / 20\%);\\stratify labels.
  \item Fit the classifier \textbf{only on training data}.
  \item Predict and evaluate on held-out test data.
\end{enumerate}
\end{frame}

\begin{frame}{A simple classifier: $k$-nearest neighbors}
\begin{itemize}
  \item $k$-NN predicts a class based on nearby training examples.
  \item Hyperparameter $k$: number of neighbors used for prediction.
  \item Start with $k=5$\\(compare with different values, such as $k=1$).
  \item High \emph{training} accuracy does not guarantee good performance on new data.
\end{itemize}
\end{frame}

\begin{frame}{Why separate training and testing?}
\begin{itemize}
  \item Evaluating on data used for fitting can be misleading.
  \item A held-out test set estimates performance on unseen examples.
  \item Keep test observations out of training and data-driven preprocessing.
  \item \textbf{Data leakage} can make test results seem better than they are.
\end{itemize}
\end{frame}

\begin{frame}{Accuracy and the confusion matrix}
\[
\text{Accuracy}=\frac{\text{Number of correct predictions}}{\text{Total predictions}}
\]
\begin{itemize}
  \item Confusion matrix:\\ \textbf{rows = true classes}, \textbf{columns = predicted classes} (scikit-learn convention).
  \item Diagonal entries: correct predictions.
  \item Off-diagonal entries: classes the model confuses.
\end{itemize}
\end{frame}

\begin{frame}{Precision, recall, and F1}
\[
\text{Precision}=\frac{TP}{TP+FP},\qquad
\text{Recall}=\frac{TP}{TP+FN}
\]
\vspace{1ex}
\[
F_1=2\times\frac{\text{Precision}\times\text{Recall}}{\text{Precision}+\text{Recall}}
\]
\begin{itemize}
  \item Precision: of predicted positives, how many are correct?
  \item Recall: of actual positives, how many are found?
  \item For multiple classes, inspect per-class results and macro averages.
\end{itemize}
\vspace{3ex}
\footnotesize{PS. F1 refers to the F-score with $\beta = 1$, meaning precision and recall are given equal importance.}
\end{frame}

\begin{frame}{Validation and model improvement}
\begin{itemize}
  \item A single train/test split is a starting point, not a guarantee.
  \item Cross-validation evaluates performance on several subsets.
  \item Tune hyperparameters with training data / cross-validation, \textbf{not} the final test set.
  \begin{itemize}
  \item Tuning hyperparameters means finding the best settings for a machine learning model to improve its performance.
  \item[]  For instance:
  \item \footnotesize{K-Nearest Neighbors $\rightarrow$ Number of neighbors used for prediction.}
  \item \footnotesize{Decision Tree $\rightarrow$ Maximum depth of the tree.}
  \item \footnotesize{Random Forest $\rightarrow$ Number of decision trees.}
  \item \footnotesize{Neural Network $\rightarrow$ How quickly the model updates its weights.}
  \end{itemize}
  \item Consider:
  \begin{itemize}
  \item class imbalance (when one class has significantly more observations than another),
  \item error costs (different types of prediction mistakes (false positive vs false negative),
  \item and data leakage (information that would not legitimately be available at prediction time is used during model training or selection).
  \end{itemize}
\end{itemize}
\end{frame}
