\begin{frame}{Two Ways to Think About Repetition}
  \begin{columns}
    \column{0.5\textwidth}
    \textbf{Iteration}
    \begin{itemize}
      \item repeats with loops;
      \item often uses \texttt{for} or \texttt{while};
      \item usually easy to trace step by step.
    \end{itemize}
    \column{0.5\textwidth}
    \textbf{Recursion}
    \begin{itemize}
      \item repeats by calling the same function again;
      \item solves a smaller version of the problem;
      \item must have a stopping rule.
    \end{itemize}
  \end{columns}

  \vspace{0.5cm}
  Both can solve the same problem.\\
  The difference is how the repeated work is organized.
\end{frame}

\begin{frame}[fragile]{Small Iterative Example: Counting Plays}
  Iteration uses a loop.

  \begin{lstlisting}
plays = [950, 870, 620]

total = 0
for play_count in plays:
    total = total + play_count

print(total)
  \end{lstlisting}

  \vspace{0.3cm}
  The loop visits each value once.

  \[
    950 + 870 + 620 = 2440
  \]
\end{frame}

\begin{frame}[fragile]{Small Recursive Example: Counting Plays}
  Recursion breaks the list into one item plus the rest.

  \begin{lstlisting}
def total_plays(values):
    if len(values) == 0:        # base case
        return 0
    return values[0] + total_plays(values[1:])

print(total_plays([950, 870, 620]))
  \end{lstlisting}

  \vspace{0.3cm}
  The function calls itself with a smaller list each time.

  \[
    [950,870,620] \rightarrow [870,620] \rightarrow [620] \rightarrow []
  \]
\end{frame}

\begin{frame}{The Most Important Rule of Recursion}
  Every recursive function needs:

  \[
    \boxed{\text{Base Case} + \text{Recursive Case}}
  \]

  \begin{center}
    \begin{tabular}{p{0.2\textwidth} p{0.7\textwidth}}
      \toprule
      Part & Meaning \\
      \midrule
      Base case & The condition that stops the function \\
      Recursive case & The part where the function calls itself on a smaller problem \\
      \bottomrule
    \end{tabular}
  \end{center}

  Without a base case, the function keeps calling itself until Python raises an error.
\end{frame}

\begin{frame}[fragile]{Iterative Sorting Idea: Neighbor Swaps}
  One simple sorting strategy compares neighboring records.

  \begin{lstlisting}
if result[j][1] < result[j + 1][1]:
    result[j], result[j + 1] = result[j + 1], result[j]
  \end{lstlisting}

  \vspace{0.3cm}
  Example comparison:

  \begin{center}
    \begin{tabular}{l r c l r}
      \toprule
      Artist A & Plays & Compare & Artist B & Plays \\
      \midrule
      Billie Eilish & 620 & $<$ & Drake & 810 \\
      \bottomrule
    \end{tabular}
  \end{center}

  Because 620 is less than 810, swap them.
\end{frame}

\begin{frame}[fragile]{Iterative Ranking Function}
  \begin{lstlisting}
def rank_iterative(data):
    result = data.copy()
    n = len(result)

    for i in range(n):
        for j in range(0, n - i - 1):
            if result[j][1] < result[j + 1][1]:
                result[j], result[j + 1] = result[j + 1], result[j]

    return result
  \end{lstlisting}

  \vspace{0.3cm}
  This uses nested loops.

  \[
    \text{Outer loop} \times \text{inner loop} \approx n \times n
  \]
\end{frame}

\begin{frame}[fragile]{Recursive Ranking Function}
  Here is one beginner-friendly recursive sorting approach.

  \begin{lstlisting}
def rank_recursive(data):
    result = data.copy()

    def bubble_pass(end):
        if end <= 1:              # base case
            return

        for j in range(end - 1):
            if result[j][1] < result[j + 1][1]:
                result[j], result[j + 1] = result[j + 1], result[j]

        bubble_pass(end - 1)      # recursive case

    bubble_pass(len(result))
    return result
  \end{lstlisting}
\end{frame}

  \begin{center}

      \toprule

      Stopping rule & Loop range ends & Base case is reached \\
      Easy for beginners? & Often yes & Sometimes less obvious \\
      Can solve sorting? & Yes & Yes \\

\begin{frame}{Why Do We Need Big-O?}
  Small examples can hide big problems.

  \begin{center}
    \begin{tabular}{r r}
      \toprule
      Number of Artists $n$ & Approximate $n^2$ Work \\
      \midrule
      10 & 100 \\
      20 & 400 \\
      50 & 2,500 \\
      100 & 10,000 \\
      1,000 & 1,000,000 \\
      \bottomrule
    \end{tabular}
  \end{center}

  \vspace{0.3cm}
  Big-O helps us describe how work grows as data grows.
\end{frame}

\begin{frame}{Common Big-O Patterns}
  \begin{center}
    \begin{tabular}{l l l}
      \toprule
      Big-O & Name & Example Idea \\
      \midrule
      $O(1)$ & Constant & Get the first item \\
      $O(n)$ & Linear & Visit every artist once \\
      $O(n \log n)$ & Efficient sorting & Python's built-in sort \\
      $O(n^2)$ & Quadratic & Compare many pairs \\
      \bottomrule
    \end{tabular}
  \end{center}

  \vspace{0.3cm}
  Big-O ignores small details and focuses on growth.
\end{frame}

\begin{frame}[fragile]{Measuring Time with \texttt{timeit}}
  Theoretical complexity is useful, but we can also measure runtime.

  \begin{lstlisting}
import timeit

iterations = 1000

iterative_time = timeit.timeit(
    lambda: rank_iterative(music_data),
    number=iterations
)

recursive_time = timeit.timeit(
    lambda: rank_recursive(music_data),
    number=iterations
)

print(iterative_time / iterations)
print(recursive_time / iterations)
  \end{lstlisting}
\end{frame}

\begin{frame}{Takeaways}
  \begin{itemize}
    \item Iteration repeats work using loops.
    \item Recursion repeats work using function calls.
    \item Recursive functions need a base case.
    \item Big-O describes growth, not exact seconds.
    \item Two algorithms can have the same Big-O but different measured runtimes.
    \item Built-in Python tools are often preferred in professional code because they are tested and optimized.
  \end{itemize}
\end{frame}
