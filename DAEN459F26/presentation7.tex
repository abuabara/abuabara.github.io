  \url{https://arxiv.org/abs/1803.09010}

  \url{https://arxiv.org/abs/1810.03993}

\begin{frame}{How to use these sources in capstone projects}
\begin{itemize}
  \item Use \textbf{NIST AI RMF} to identify and govern AI-related risks.
  \item Use \textbf{NIST Privacy Framework} and \textbf{GDPR principles} to justify privacy controls.
  \item Use \textbf{OWASP Top 10} to evaluate application and API security.
  \item Use \textbf{FAIR principles} to improve data stewardship.
  \item Use \textbf{Datasheets} and \textbf{Model Cards} as required documentation templates.
\end{itemize}
\end{frame}

\begin{frame}{Why responsible data engineering matters}
\begin{itemize}
  \item Data pipelines create real-world consequences.
  \item Ethical failures often begin as design shortcuts.
  \item Responsible systems require technical, legal, and social controls.
\end{itemize}
\end{frame}

\begin{frame}{Core responsibilities}
\begin{itemize}
  \item Collect only what is needed.
  \item Protect data across its lifecycle.
  \item Document provenance, consent, assumptions, and limitations.
  \item Test for bias, leakage, misuse, and downstream harm.
  \item Make systems auditable and governable.
\end{itemize}
\end{frame}

\begin{frame}{Privacy}
\begin{itemize}
  \item Data minimization
  \item Purpose limitation
  \item Consent and notice
  \item De-identification and re-identification risk
  \item Retention and deletion policies
\end{itemize}
\end{frame}

\begin{frame}{Security}
\begin{itemize}
  \item Authentication and authorization
  \item Encryption in transit and at rest
  \item Secrets management
  \item Least privilege
  \item Logging, monitoring, and incident response
\end{itemize}
\end{frame}

\begin{frame}{Bias and fairness}
\begin{itemize}
  \item Historical bias
  \item Sampling bias
  \item Label bias
  \item Measurement bias
  \item Deployment feedback loops
\end{itemize}
\end{frame}

\begin{frame}{Responsible data lifecycle}
\begin{enumerate}
  \item Define purpose and stakeholders.
  \item Assess data sources and permissions.
  \item Build secure pipelines.
  \item Validate quality and representativeness.
  \item Document assumptions and risks.
  \item Monitor performance and harms after deployment.
\end{enumerate}
\end{frame}

\begin{frame}{Project risk checklist}
\begin{itemize}
  \item Who could be harmed by errors?
  \item What sensitive attributes are present or inferable?
  \item Could the data be re-identified?
  \item Who has access to raw data?
  \item Are outputs explainable and contestable?
  \item What happens when the model or pipeline fails?
\end{itemize}
\end{frame}

\begin{frame}{Documentation artifacts}
\begin{itemize}
  \item Data dictionary
  \item Data lineage diagram
  \item Dataset datasheet
  \item Security threat model
  \item Bias and fairness assessment
  \item Retention and deletion policy
  \item Model card, if ML is used
\end{itemize}
\end{frame}
