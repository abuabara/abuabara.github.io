% Slide 2: The Core Workflow
\begin{frame}{The Discovery Workflow}
    \begin{itemize}
        \item \textbf{Problem Framing:} Defining the right challenge.
        \item \textbf{Needs Assessment:} Identifying the foundational gaps.
        \item \textbf{Requirements Elicitation:} Gathering specific technical rules.
    \end{itemize}
\end{frame}

% Slide 3: Problem Framing
\begin{frame}{1. Problem Framing}
    \textbf{Concept:} Shifting from surface-level symptoms to root business challenges. Avoids building expensive pipelines for the wrong issue.
    
    \vspace{0.3cm}
    \textbf{Example:}
    \begin{itemize}
        \item \emph{Symptom:}\\"The dashboard is too slow."
        \item \emph{Poor Framing:}\\"We need to rewrite the entire ETL pipeline in Spark."
        \item \emph{Good Framing:}\\"Business analysts lose decision-making time because downstream BI tables lack optimized partitioning and indexing for morning reports."
    \end{itemize}
\end{frame}

% Slide 4: Needs Assessment
\begin{frame}{2. Needs Assessment}
    \textbf{Concept:} Evaluating the current state vs. the desired future state. Separates user "wants" from systemic "needs."
    
    \vspace{0.3cm}
    \textbf{Example:}
    \begin{itemize}
        \item \emph{User Want:}\\"We want real-time streaming data for all sales metrics."
        \item \emph{Assessment Finding:}\\The finance team only reviews reports weekly. The underlying source systems update via daily batch jobs. 
        \item \emph{The True Need:}\\A robust, automated daily batch pipeline with data quality checks, rather than an expensive Kafka infrastructure.
    \end{itemize}
\end{frame}

% Slide 5: Requirements Elicitation
\begin{frame}{3. Requirements Elicitation}
    \textbf{Concept:} Actively extracting detailed, actionable constraints (functional and non-functional) from stakeholders and systems.
    
    \vspace{0.3cm}
    \textbf{Example:}
    \begin{itemize}
        \item \textbf{Functional:}\\The pipeline must ingest JSON files from an S3 bucket, schema-validate them, and flatten them into a Snowflake warehouse.
        \item \textbf{Non-Functional (SLA):}\\Data must be available by 6:00 AM UTC daily; the pipeline must handle up to 500\% traffic spikes during Black Friday.
    \end{itemize}
\end{frame}

% Slide 6: Summary
\begin{frame}{Summary Checklist for Teams}
    \begin{enumerate}
        \item \hl{Don't write code} during \textbf{Problem Framing}.\\Focus entirely on the {\it business} bottleneck.
        \item Audit your sources (thoroughly inspecting the data systems you depend on) and destinations during \textbf{Needs Assessment}. 
        \item \ul{Lock down metrics, volume, and latency} during \textbf{Requirements Elicitation}.
    \end{enumerate}
\end{frame}
