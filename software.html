  • Dock, show/hide speed:
    defaults write com.apple.dock autohide-time-modifier -int 0
    defaults write com.apple.dock autohide-delay -float 0
  • Stage Manager, show/hide speed:
    defaults write com.apple.WindowManager StageFrameMinimumHorizontalInset -int 0
    defaults write com.apple.WindowManager AutoHideDelay -int 0
    defaults write com.apple.WindowManager AnimationSpeed -int 65536
  • Menubar, reduce spacing between items:
    defaults -currentHost write -globalDomain NSStatusItemSelectionPadding -int 7
    defaults -currentHost write -globalDomain NSStatusItemSpacing -int 7
  • Screenshots, change format and no shadow:
    defaults write com.apple.screencapture type jpg
    defaults write com.apple.screencapture disable-shadow -bool TRUE

  • openssl
    download
    https://openssl-library.org/source/
    or
    curl -O https://github.com/openssl/openssl/releases/download/openssl-3.5.4/openssl-3.5.4.tar.gz
    unzip
    tar -xf openssl-3.5.4.tar.gz
    then
    cd openssl-3.5.4
    mkdir ~/Library/wget
    ./configure --prefix=$HOME/Library/wget
    make
    make install

    wget
    curl -O https://ftp.gnu.org/gnu/wget/wget-1.25.0.tar.gz
    tar -xf wget-1.25.0.tar.gz
    cd wget-1.25.0
    ./configure --prefix=/usr/local
    make
    make install
    wget --version


  • graphviz
    # find the link from https://gitlab.com/graphviz/graphviz/-/releases (don't use Assets—use “Other”)
    wget https://gitlab.com/api/v4/projects/4207231/packages/generic/graphviz-releases/14.0.4/graphviz-14.0.4.tar.gz
    # unzip the source code
    tar -xf graphviz-14.0.4.tar
    cd graphviz-14.0.4
    # create a folder to save graphviz locally
    mkdir ~/Library/graphviz
    ./configure --prefix=$HOME/Library/graphviz
    make
    make install
    # Check the installation
    dot -V

    # MacPorts
    export PATH="/opt/local/bin:/opt/local/sbin:$PATH"
    # Graphviz

    export LDFLAGS="$HOME/Library/wget/lib"
